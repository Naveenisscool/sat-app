// Base Static Question Repository (College Board Aligned)
const STATIC_BANK = [
  {
    subject: "rw", topic: "Transitions",
    q: "The initial clinical trials yielded inconclusive results. ________, researchers redesigned the testing apparatus to eliminate external atmospheric variables.",
    a: ["Furthermore", "Therefore", "However", "In contrast"], c: 1,
    e: "'Therefore' accurately establishes a cause-and-effect connection between inconclusive results and modifying test equipment."
  },
  {
    subject: "rw", topic: "Boundaries & Punctuation",
    q: "Which choice completes the sentence with correct standard punctuation?<br><br>The astronomer identified three distinct exoplanets________ each orbiting a distant red dwarf star.",
    a: [";", ",", ":", "---"], c: 2,
    e: "A colon introduces an explanatory expansion of the main clause that immediately precedes it."
  },
  {
    subject: "rw", topic: "Words in Context",
    q: "The lead investigator described the empirical evidence as ________ because it decisively resolved a debate that had persisted for decades.",
    a: ["tentative", "equivocal", "definitive", "superficial"], c: 2,
    e: "'Definitive' means conclusive and resolving all uncertainty."
  },
  {
    subject: "rw", topic: "Subject-Verb Agreement",
    q: "A substantial collection of rare manuscripts ________ stored securely in the climate-controlled vaults.",
    a: ["are", "were", "is", "have been"], c: 2,
    e: "The singular subject noun 'collection' requires the singular verb 'is'."
  },
  {
    subject: "math", topic: "Lines & Triangles",
    q: "In right triangle ABC, angle C is 90°. If sin(A) = 5/13, what is the value of cos(B)?",
    a: ["5/13", "12/13", "13/5", "12/5"], c: 0,
    e: "By cofunction identity, sin(A) = cos(90° - A) = cos(B) = 5/13."
  }
];

// Dynamic Question Generator Engine (Scales to 300+ unique mathematical permutations)
function generateDynamicMath() {
  const generators = [
    // 1. Linear Systems
    () => {
      let x = Math.floor(Math.random() * 10) + 2;
      let y = Math.floor(Math.random() * 8) + 1;
      let a = Math.floor(Math.random() * 4) + 2;
      let eq1Val = a * x + y;
      let eq2Val = x - y;
      return {
        subject: "math", topic: "Systems of Linear Equations",
        q: `In the system of equations below, what is the value of x?<br><br>${a}x + y = ${eq1Val}<br>x - y = ${eq2Val}`,
        a: [`x = ${x}`, `x = ${x + 2}`, `x = ${x - 1}`, `x = ${x * 2}`], c: 0,
        e: `Add both equations together: (${a}x + x) + (y - y) = ${eq1Val} + ${eq2Val} => ${a+1}x = ${eq1Val + eq2Val} => x = ${x}.`
      };
    },
    // 2. Quadratics in Vertex Form
    () => {
      let h = Math.floor(Math.random() * 8) + 1;
      let k = Math.floor(Math.random() * 15) + 3;
      return {
        subject: "math", topic: "Quadratic Functions",
        q: `The function f(x) = (x - ${h})^2 + ${k} models a parabola in the xy-plane. What is the minimum value of f(x)?`,
        a: [`${k}`, `${h}`, `-${h}`, `-${k}`], c: 0,
        e: `In vertex form f(x) = a(x - h)^2 + k, the vertex is (h, k) = (${h}, ${k}). The minimum y-value is k = ${k}.`
      };
    },
    // 3. Circle Equations
    () => {
      let h = Math.floor(Math.random() * 6) + 1;
      let k = Math.floor(Math.random() * 6) + 1;
      let r = Math.floor(Math.random() * 7) + 3;
      let r2 = r * r;
      return {
        subject: "math", topic: "Circle Equations",
        q: `A circle in the xy-plane is given by (x + ${h})^2 + (y - ${k})^2 = ${r2}. What are its center coordinates and radius?`,
        a: [`Center (-${h}, ${k}), Radius = ${r}`, `Center (${h}, -${k}), Radius = ${r}`, `Center (-${h}, ${k}), Radius = ${r2}`, `Center (${h}, ${k}), Radius = ${r}`], c: 0,
        e: `In standard form (x - h)^2 + (y - k)^2 = r^2, center is (-${h}, ${k}) and radius is √${r2} = ${r}.`
      };
    },
    // 4. Exponential Growth
    () => {
      let p0 = (Math.floor(Math.random() * 5) + 1) * 100;
      let hrs = Math.floor(Math.random() * 4) + 2;
      return {
        subject: "math", topic: "Exponential Growth",
        q: `A bacterial culture contains ${p0} organisms and doubles in size every ${hrs} hours. Which equation models population P after t hours?`,
        a: [`P = ${p0}(2)^(t/${hrs})`, `P = ${p0}(2)^(${hrs}t)`, `P = ${p0}(${hrs})^(t/2)`, `P = ${p0} + 2^(t/${hrs})`], c: 0,
        e: `Exponential growth formula P = P0(2)^(t / period). Here P0 = ${p0} and period = ${hrs} hours.`
      };
    },
    // 5. Linear Equation Solving
    () => {
      let mult = Math.floor(Math.random() * 4) + 2;
      let sub = Math.floor(Math.random() * 6) + 2;
      let ans = Math.floor(Math.random() * 10) + 3;
      let rhs = mult * (ans - sub);
      return {
        subject: "math", topic: "Linear Equations in 1 Variable",
        q: `Solve for x: ${mult}(x - ${sub}) = ${rhs}`,
        a: [`x = ${ans}`, `x = ${ans + 3}`, `x = ${ans - 2}`, `x = ${ans * 2}`], c: 0,
        e: `Divide both sides by ${mult}: x - ${sub} = ${rhs / mult}. Add ${sub}: x = ${ans}.`
      };
    }
  ];

  let chosenGen = generators[Math.floor(Math.random() * generators.length)];
  return chosenGen();
}

// Generate pool containing guaranteed unique variations
function getQuestionPool(mode) {
  let list = [];
  if (mode === 'rw') {
    list = STATIC_BANK.filter(q => q.subject === 'rw');
  } else if (mode === 'math') {
    list = STATIC_BANK.filter(q => q.subject === 'math');
    for (let i = 0; i < 20; i++) list.push(generateDynamicMath());
  } else {
    list = [...STATIC_BANK];
    for (let i = 0; i < 25; i++) list.push(generateDynamicMath());
  }
  return list.sort(() => 0.5 - Math.random());
}

let state = {
  xp: parseInt(localStorage.getItem('sat_priv_xp') || '0'),
  total: parseInt(localStorage.getItem('sat_priv_total') || '0'),
  correct: parseInt(localStorage.getItem('sat_priv_correct') || '0'),
  round: [],
  idx: 0,
  selected: null
};

let desmosInstance = null;

function el(id) { return document.getElementById(id); }

function updateXP() {
  el('xp').innerText = state.xp;
  localStorage.setItem('sat_priv_xp', state.xp);
  localStorage.setItem('sat_priv_total', state.total);
  localStorage.setItem('sat_priv_correct', state.correct);
}

function startRound(mode) {
  state.round = getQuestionPool(mode).slice(0, 5);
  state.idx = 0;
  el('home').classList.add('hidden');
  el('game').classList.remove('hidden');
  renderQ();
}

function renderQ() {
  state.selected = null;
  const q = state.round[state.idx];
  el('qTag').innerText = q.topic.toUpperCase();
  el('qNum').innerText = `Question ${state.idx + 1} of ${state.round.length}`;
  el('progress').style.width = `${((state.idx) / state.round.length) * 100}%`;
  el('qText').innerHTML = q.q;
  el('explainBox').innerHTML = '';
  el('nextBtn').disabled = true;
  el('nextBtn').innerText = "Check Answer";

  // Create scrambled copy of answers
  let answers = q.a.map((opt, i) => ({ text: opt, isCorrect: i === q.c })).sort(() => 0.5 - Math.random());
  q.currentAnswers = answers;

  el('answersBox').innerHTML = answers.map((item, i) => `
    <div class="answer" id="ans_${i}" onclick="selectAns(${i})">
      <strong>${String.fromCharCode(65 + i)}</strong>
      <div>${item.text}</div>
    </div>
  `).join('');
}

function selectAns(i) {
  if (el('nextBtn').innerText === "Next Question") return;
  state.selected = i;
  document.querySelectorAll('.answer').forEach((a, idx) => a.classList.toggle('selected', idx === i));
  el('nextBtn').disabled = false;
}

function checkOrNext() {
  if (el('nextBtn').innerText === "Next Question") {
    state.idx++;
    if (state.idx < state.round.length) renderQ();
    else goHome();
    return;
  }

  const q = state.round[state.idx];
  const chosen = q.currentAnswers[state.selected];
  const correctIdx = q.currentAnswers.findIndex(a => a.isCorrect);
  
  state.total++;

  if (chosen.isCorrect) {
    state.correct++;
    state.xp += 20;
    el(`ans_${state.selected}`).classList.add('correct');
  } else {
    el(`ans_${state.selected}`).classList.add('wrong');
    el(`ans_${correctIdx}`).classList.add('correct');
  }

  updateXP();
  el('explainBox').innerHTML = `
    <div class="explain">
      <strong>${chosen.isCorrect ? "✅ Correct!" : "❌ Incorrect"}</strong><br>${q.e}
    </div>
  `;
  el('nextBtn').innerText = "Next Question";
}

function toggleDesmos() {
  const wrap = el('desmosWrap');
  wrap.classList.toggle('hidden');
  
  if (!wrap.classList.contains('hidden')) {
    if (!desmosInstance && typeof Desmos !== 'undefined') {
      desmosInstance = Desmos.GraphingCalculator(el('calculator'), { autosize: true });
    } else if (desmosInstance) {
      desmosInstance.resize();
    }
  }
}

function goHome() {
  el('game').classList.add('hidden');
  el('dashboard').classList.add('hidden');
  el('home').classList.remove('hidden');
  updateXP();
}

function showDashboard() {
  el('home').classList.add('hidden');
  el('dashboard').classList.remove('hidden');
  el('statTotal').innerText = state.total;
  const acc = state.total > 0 ? Math.round((state.correct / state.total) * 100) : 0;
  el('statAcc').innerText = acc + '%';
}

updateXP();
